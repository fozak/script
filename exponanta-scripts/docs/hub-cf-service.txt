worker.js — two universal OAuth routes
javascript
// ── OAuth redirect — any provider ────────────────────────
if (path.startsWith('/auth/') && path.endsWith('/login')) {
  const provider   = path.split('/')[2]
  const oauthCfg   = CW._config.oauth[provider]
  if (!oauthCfg) return new Response('Unknown provider', { status: 404 })

  const return_url = url.searchParams.get('return_url') || '/'
  const params     = new URLSearchParams({
    client_id:     env[`${provider.toUpperCase()}_CLIENT_ID`],
    redirect_uri:  `${env.WORKER_URL}/auth/${provider}/callback`,
    response_type: 'code',
    scope:         oauthCfg.scope,
    state:         btoa(return_url),
  })
  return Response.redirect(`${oauthCfg.authUrl}?${params}`, 302)
}

// ── OAuth callback — any provider ────────────────────────
if (path.startsWith('/auth/') && path.endsWith('/callback')) {
  const provider   = path.split('/')[2]
  const code       = url.searchParams.get('code')
  const return_url = atob(url.searchParams.get('state') || btoa('/'))

  const run_doc = {
    operation:      'loginWithOAuth',
    target_doctype: 'User',
    input:          { provider, code },
    user:           {}, target: null, query: {}, options: {}
  }

  await CW.controller(run_doc)

  if (!run_doc.success)
    return Response.redirect(`${return_url}?error=${encodeURIComponent(run_doc.error)}`, 302)

  return Response.redirect(`${return_url}#token=${run_doc.user.token}`, 302)
}

Client — trigger from any third-party site
javascript
function oauthLogin(provider, returnUrl = window.location.href) {
  window.location.href =
    `${CW._config.hub.url}auth/${provider}/login?return_url=${encodeURIComponent(returnUrl)}`
}

oauthLogin('linkedin')  // → redirects through hub → LinkedIn → back to caller page

Token extraction on return (on any page that receives the redirect back):

javascript
const token = new URLSearchParams(window.location.hash.slice(1)).get('token')
if (token) {
  const payload = JSON.parse(atob(token.split('.')[1]))
  const currentUser = { ...payload, token }
  localStorage.setItem('currentUser', JSON.stringify(currentUser))
  globalThis.currentUser = currentUser
  window.history.replaceState({}, '', window.location.pathname)
}

Key notes from that session:

Token passes back in #fragment (not ?query) — browsers strip query params on cross-origin 302 redirects, fragments survive
LinkedIn's callback must be server-side (hub Worker handles it) — unlike Google which can be client-side
callbackPath in config is redundant — Worker derives it from provider name: /auth/${provider}/callback
Cloudflare secrets needed: LINKEDIN_CLIENT_ID, LINKEDIN_CLIENT_SECRET, WORKER_URL
Redirect URI registered in LinkedIn Developer Portal: https://hub-cf.i771468.workers.dev/auth/linkedin/callback
