The bisync decision function — the core:

js
function bisyncDecision(loc, rem) {
  // identical — skip
  if (loc.size === rem.size && loc.modified === rem.modified) return 'skip';

  // size differs — newer modtime wins
  if (loc.size !== rem.size) {
    return loc.modified > rem.modified ? 'upload' : 'download';
  }

  // size same, modtime differs — D1 edited content wins
  if (rem.content) return 'download';
  return 'upload';
}

The snapshot fast-check — runs on folder-reader open:

js
async function isDirty(snapshot) {
  for (const s of snapshot) {
    const id  = generateId('File', s.path);
    const rec = globalThis._fsIndex.get(id);
    if (!rec) return true;                              // deleted locally
    if (rec.data.size !== s.size) return true;          // size changed
    if (rec.data.modified !== s.modified) return true;  // touched
  }
  // check for new local files not in snapshot
  for (const rec of globalThis._fsIndex.values()) {
    if (!snapshot.find(s => s.path === rec.data.path)) return true;
  }
  return false;
}

Early exit on first dirty file — O(1) best case, O(n) worst case.
